def dparse(s):
    # dd.mm or yyyy-mm-dd
    if isinstance(s,date): return s
    if re.match(r"\d{4}-\d{2}-\d{2}$",s):
        return datetime.strptime(s,"%Y-%m-%d").date()
    return datetime.strptime(s+".2026","%d.%m.%Y").date()

def weekly(start,end,weekday):
    s,e=dparse(start),dparse(end)
    out=[]
    cur=s
    while cur.weekday()!=weekday:
        cur+=timedelta(days=1)
    while cur<=e:
        if cur not in HOLIDAYS:
            out.append(cur)
        cur+=timedelta(days=7)
    return out

def explicit(ds):
    return [dparse(x) for x in ds]

def norm_time(t):
    h,m = map(int,t.split(":"))
    return f"{h:02d}:{m:02d}"

def event_desc(title, segments, extra=None):
    parts=[]
    if len(segments)>1:
        breaks=[]
        for (_,end1),(start2,_) in zip(segments,segments[1:]):
            if end1!=start2:
                breaks.append(f"{norm_time(end1)}–{norm_time(start2)}")
        if breaks:
            parts.append("Перерыв: "+", ".join(breaks)+".")
    if title in attestation:
        parts.append(f"Форма промежуточной аттестации: {attestation[title]}.")
    if extra:
        parts.append(extra.strip())
    return " ".join(parts)

def add(gs,title,dates,segments,location=None,kind="lesson",sourceCell=None,sourceRange=None,extra_desc=None):
    if isinstance(gs,str): gs=[gs]
    if isinstance(dates,date): dates=[dates]
    if isinstance(dates,str): dates=[dparse(dates)]
    segs=[tuple(x) for x in segments]
    st=norm_time(segs[0][0]); en=norm_time(segs[-1][1])
    loc = location if location is not None else locations.get(title,"")
    desc=event_desc(title,segs,extra_desc)
    for g in gs:
        for dt in dates:
            events[g].append({
                "title": title,
                "start": f"{dt.isoformat()}T{st}:00+03:00",
                "end": f"{dt.isoformat()}T{en}:00+03:00",
                "location": loc,
                "kind": kind,
                **({"description":desc} if desc else {}),
                **({"sourceCell":sourceCell} if sourceCell else {}),
                **({"sourceRange":sourceRange} if sourceRange else {}),
            })
